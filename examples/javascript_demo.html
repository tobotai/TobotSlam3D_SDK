# javascript_demo.html 完整示例 — TOBOT AI SDK # 依赖：见 README # 更多说明见 README.md 与 docs/ # 官网完整版：https://www.tobotai.com/sdk/index.html const ws = new WebSocket('ws://192.168.1.151:9090'); ws.onopen = () => { // Subscribe to all 5 Topics sub('/fast_lio_localization','nav_msgs/Odometry'); sub('/aft_pgo_path','nav_msgs/Path'); sub('/map_list_kitty','kitty_runtime_manager/MapList'); sub('/ebot_goal_status_local_planner','std_msgs/Float64MultiArray'); sub('/ebot_work_state','std_msgs/Float64MultiArray'); // Call all 9 Services svc('/start_mapping_kitty', {}); svc('/save_map_kitty', {filepath:'test_map'}); svc('/select_map_and_start_kitty', {filepath:'test_map'}); svc('/navigate_to_pose', {goal:{pose:{position:{x:1,y:2,z:0}}}}); svc('/navigate_to_pose', {goal:{pose:{position:{x:999999,y:888888,z:0}}}}); svc('/add_goal_kitty', {map_name:'test_map',goal_name:'Reception',x:1,y:2,theta:0.5}); svc('/delete_goal_kitty', {map_name:'test_map',goal_name:'Reception'}); svc('/docking_kitty', {}); svc('/delete_map_kitty', {map_name:'test_map'}); }; function sub(t,type){ws.send(JSON.stringify({op:'subscribe',topic:t,type}));} function svc(n,args){ws.send(JSON.stringify({op:'call_service',service:n,args}));} ws.onmessage = e => { const m = JSON.parse(e.data); if (m.op==='publish') { if (m.topic==='/ebot_work_state') console.log('[Battery]', m.msg.data[1]+'%', m.msg.data[0]===1?'充电':'Not charging'); else if (m.topic==='/ebot_goal_status_local_planner' && m.msg.data[0]===1) console.log('✅ Arrived'); else if (m.topic==='/fast_lio_localization') { const p = m.msg.pose.pose.position; console.log('[Pose]', p.x.toFixed(2), p.y.toFixed(2)); } } if (m.op==='service_response') console.log('[SVC]', m.service, '→', m.values); };
